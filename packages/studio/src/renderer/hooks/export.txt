  packages/studio/
  ├── Electron 主进程 (src/main/)
  │   ├── index.ts, preload.ts
  │   └── ipc/ (init, pipeline, state, config)
  ├── React 渲染进程 (src/renderer/)
  │   ├── modules/ (studio, dashboard, monitor, settings, analytics)
  │   ├── components/ui/ (8个Radix组件)
  │   ├── hooks/ (6个业务hooks)
  │   └── lib/ (api, ws, constants)
  └── shared/types.ts (完整IPC类型定义)
